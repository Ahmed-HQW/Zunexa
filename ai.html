يمكنك تشغيل Zunexa Archon CLI مباشرة من مبادل الأوامر (CMD/PowerShell/Terminal/Termux) للتفاعل مع الوكلاء والذكاء الاصطناعي من سطر الأوامر. 

// install: npm install -g axios
const axios = require('axios');
const readline = require('readline');

const rl = readline.createInterface({ input: process.stdin, output: process.stdout });
console.log("\x1b[33m%s\x1b[0m", "--- ZUNEXA ARCHON CLI V13 ---");

function ask() {
  rl.question("\x1b[36mArchon> \x1b[0m", async (q) => {
    if (q.trim() === 'exit') process.exit();
    try {
      const res = await axios.post('https://text.pollinations.ai/openai', {
        messages: [{ role: 'system', content: 'You are Zunexa Archon AI CLI' }, { role: 'user', content: q }]
      });
      console.log("\n" + res.data.choices[0].message.content + "\n");
    } catch (e) { console.log("Error connecting to Archon Engine."); }
    ask();
  });
}
ask();
# install: pip install requests
import requests, sys

print("\033[93m--- ZUNEXA ARCHON CLI (Python) ---\033[0m")
while True:
    try:
        q = input("\033[96mArchon> \033[0m")
        if q.strip() == "exit": break
        res = requests.post('https://text.pollinations.ai/openai', json={
            "messages": [{"role": "system", "content": "You are Zunexa Archon AI CLI"}, {"role": "user", "content": q}]
        }).json()
        print("\n" + res['choices'][0]['message']['content'] + "\n")
    except KeyboardInterrupt: sys.exit()
    except Exception as e: print("Error connecting to Archon Engine.")
